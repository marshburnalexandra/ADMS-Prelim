-- 1. Total revenue by branch city
SELECT dim_branch.branch_city,
	SUM(fact_sales.quantity * dim_product.price) AS total_revenue
FROM fact_sales
JOIN dim_branch
	ON fact_sales.branch_id = dim_branch.branch_id
JOIN dim_product
	ON fact_sales.product_id = dim_product.product_id
GROUP BY dim_branch.branch_city;

-- 2. Top 5 customers who spends the most
SELECT dim_customer.customer_name,
	SUM(fact_sales.quantity * dim_product.price) AS total_spent
FROM fact_sales
JOIN dim_customer
	ON fact_sales.customer_id = dim_customer.customer_id
JOIN dim_product
	ON fact_sales.product_id = dim_product.product_id
GROUP BY dim_customer.customer_name
ORDER BY total_spent DESC
LIMIT 5;

-- 3. Total quantity 
SELECT dim_product.product_line,
	SUM(fact_sales.quantity) AS total_quantity
FROM fact_sales
JOIN dim_product
	ON fact_sales.product_id = dim_product.product_id
GROUP BY dim_product.product_line;

-- 4. Show the distribution of payment type used
SELECT dim_payment.payment_type,
	COUNT(*) AS total_used
FROM fact_sales
JOIN dim_payment
	ON fact_sales.payment = dim_payment.payment_id
GROUP BY dim_payment.payment_type;

-- 5. Average rating per product line
SELECT dim_product.product_line,
	AVG(fact_sales.rating) AS average_rating
FROM fact_sales
JOIN dim_product
	ON fact_sales.product_id = dim_product.product_id
GROUP BY dim_product.product_line;
